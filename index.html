int isPalindrome(int number) {
    if (number < 0) return 0;
    int original = number, reversed = 0;
    while (number > 0) {

    if (c == '*' || c == '/' || c == '%') return 2;
    if (c == '+' || c == '-') return 1;

        char c = infix[i];

#include <string.h>
#include <ctype.h>
#include <math.h>
#define MAX 300

int isOperator(char c) { return c == '+' || c == '-' || c == '*' || c == '/' || c == '^'; }

int main(void)
{
    char exp[MAX];
    double stack[MAX], a, b;
    int top = -1, i, len;

    printf("Enter prefix expression (single-digit non-negative operands, no spaces): ");
    if (scanf("%299s", exp) != 1) return 1;
    len = (int)strlen(exp);

    for (i = len - 1; i >= 0; i--) {
        char c = exp[i];
        if (isdigit((unsigned char)c)) stack[++top] = c - '0';

                stack[++top] = a / b;
            } else stack[++top] = pow(a, b);
        } else { printf("Invalid character.\n"); return 1; }

    int data;
    struct Node *l, *r;
} Node;

Node *newNode(int x) {
    Node *n = malloc(sizeof(Node));
    n->data = x;
    n->l = n->r = NULL;
    return n;
}

void level(Node *r) {
    if (!r)
        return;
    Node *q[1000];
    int f = 0, b = 0;
    q[b++] = r;
    while (f < b) {
        Node *t = q[f++];
        printf("%d ", t->data);
        if (t->l)
            q[b++] = t->l;
        if (t->r)
            q[b++] = t->r;
    }
}

#include <stdio.h>
#include <stdlib.h>

typedef struct Node {
    int data;
    struct Node *l, *r;
} Node;

Node *insert(Node *n, int x) {
    if (!n) {
        n = malloc(sizeof(Node));
        n->data = x;
        n->l = n->r = NULL;
    } else if (x < n->data)
        n->l = insert(n->l, x);
    else if (x > n->data)
        n->r = insert(n->r, x);
    return n;
}

int count(Node *n) {
    return n ? 1 + count(n->l) + count(n->r) : 0;
}

void inorder(Node *n) {
    if (n) {
        inorder(n->l);
        printf("%d ", n->data);
        inorder(n->r);
    }
}

int main() {
    Node *r = NULL;
    int n, x;
    printf("Number of values: ");
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        scanf("%d", &x);
        r = insert(r, x);
    }
    printf("BST inorder: ");
    inorder(r);
    printf("\nNumber of nodes: %d\n", count(r));
    return 0;
}
